Create `analytics-dashboard-helpers.js`:

```js
(function (root) {
  "use strict";

  function reviewRequestMetric(rows, platform, queryFailed) {
    if (queryFailed) return { available: false, value: null };
    const value = (Array.isArray(rows) ? rows : [])
      .filter(row => platform === "all" || row.platform === platform)
      .reduce((total, row) => total + (Number(row.requests) || 0), 0);
    return { available: true, value };
  }

  root.AnalyticsDashboardHelpers = Object.freeze({ reviewRequestMetric });
  if (typeof module !== "undefined" && module.exports) {
    module.exports = { reviewRequestMetric };
  }
})(typeof globalThis !== "undefined" ? globalThis : this);
```

import XCTest
@testable import PrintAssistant

final class OutputLocalizationTests: XCTestCase {
    private let keys: Set<String> = [
        "output.loading_profile", "output.detail_level", "output.simple", "output.advanced",
        "output.filament_settings", "output.print_settings", "output.back_goals", "output.reset",
        "output.export.bambu", "output.export.orca", "output.export.prusa",
        "output.export.unavailable", "output.export.profile_text",
        "output.export.no_preset_note", "output.export.process_only_note",
        "output.export.unavailable_message", "output.export.write_failed", "output.export.failed",
        "output.invalid.title", "output.invalid.message", "output.invalid.cta",
        "output.ok", "output.report_issue",
    ]

    func testEnglishAndDanishExposeTheExactOutputContract() throws {
        let en = try table("en")
        let da = try table("da")
        XCTAssertEqual(Set(en.keys), keys)
        XCTAssertEqual(Set(da.keys), keys)
        XCTAssertTrue(en.values.allSatisfy { !$0.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty })
        XCTAssertTrue(da.values.allSatisfy { !$0.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty })
    }

    func testFormattedKeysHaveMatchingPlaceholderSignatures() throws {
        let en = try table("en")
        let da = try table("da")
        for key in ["output.export.no_preset_note", "output.export.process_only_note"] {
            XCTAssertEqual(en[key]?.components(separatedBy: "%@").count,
                           da[key]?.components(separatedBy: "%@").count)
            XCTAssertEqual(en[key]?.components(separatedBy: "%@").count, 2)
        }
    }

    func testUnsupportedLanguageFallsBackToEnglishDevelopmentRegion() {
        XCTAssertEqual(Bundle.main.object(forInfoDictionaryKey: "CFBundleDevelopmentRegion") as? String, "en")
        XCTAssertEqual(Bundle.preferredLocalizations(from: ["en", "da"], forPreferences: ["de"]).first, "en")
    }

    private func table(_ language: String) throws -> [String: String] {
        let path = try XCTUnwrap(Bundle.main.path(forResource: language, ofType: "lproj"))
        let bundle = try XCTUnwrap(Bundle(path: path))
        let url = try XCTUnwrap(bundle.url(forResource: "Localizable", withExtension: "strings"))
        let data = try Data(contentsOf: url)
        return try XCTUnwrap(PropertyListSerialization.propertyList(from: data, format: nil) as? [String: String])
    }
}
